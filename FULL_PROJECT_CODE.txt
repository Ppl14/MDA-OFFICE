export function formatINR(amount: number): string {
  return new Intl.NumberFormat('en-IN', {
    style: 'currency',
    currency: 'INR',
    maximumFractionDigits: 0,
  }).format(amount);
}

export function formatDate(dateStr: string | null): string {
  if (!dateStr) return '—';
  const date = new Date(dateStr);
  return date.toLocaleDateString('en-IN', {
    day: '2-digit',
    month: 'short',
    year: 'numeric',
  });
}

export function formatDateTime(dateStr: string): string {
  const date = new Date(dateStr);
  return date.toLocaleDateString('en-IN', {
    day: '2-digit',
    month: 'short',
    year: 'numeric',
  }) + ', ' + date.toLocaleTimeString('en-IN', {
    hour: '2-digit',
    minute: '2-digit',
  });
}

export function formatDateShort(dateStr: string): string {
  const date = new Date(dateStr);
  return date.toLocaleDateString('en-IN', {
    day: '2-digit',
    month: 'short',
  });
}

export function daysUntil(dateStr: string): number {
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const target = new Date(dateStr);
  target.setHours(0, 0, 0, 0);
  const diff = target.getTime() - today.getTime();
  return Math.ceil(diff / (1000 * 60 * 60 * 24));
}

export function isToday(dateStr: string): boolean {
  return daysUntil(dateStr) === 0;
}

export function isOverdue(dateStr: string): boolean {
  return daysUntil(dateStr) < 0;
}

export function isSameDay(a: string | Date, b: string | Date): boolean {
  const da = typeof a === 'string' ? new Date(a) : a;
  const db = typeof b === 'string' ? new Date(b) : b;
  return (
    da.getFullYear() === db.getFullYear() &&
    da.getMonth() === db.getMonth() &&
    da.getDate() === db.getDate()
  );
}

export function toISODate(date: Date): string {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, '0');
  const d = String(date.getDate()).padStart(2, '0');
  return `${y}-${m}-${d}`;
}

import type { PricingInput, PricingResult, PackageTier, SharingType, Batch } from '@/types';

export const PACKAGE_TIERS: PackageTier[] = ['Land-to-Land', 'One-Way Travel', 'Full Package', 'Custom'];
export const SHARING_TYPES: SharingType[] = ['Couple/Double', 'Triple', 'Quad', 'Single Private'];

export function getBasePriceForTier(batch: Batch, tier: PackageTier): number {
  switch (tier) {
    case 'Land-to-Land': return batch.base_land_price || batch.base_fare || 0;
    case 'One-Way Travel': return batch.base_train_price || 0;
    case 'Full Package': return batch.base_full_price || 0;
    case 'Custom': return 0;
    default: return 0;
  }
}

export function getSharingSurcharge(batch: Batch, sharing: SharingType): number {
  switch (sharing) {
    case 'Couple/Double': return batch.couple_surcharge || 0;
    case 'Single Private': return batch.single_supplement || 0;
    default: return 0;
  }
}

export function computeRackRate(batch: Batch, tier: PackageTier, sharing: SharingType): number {
  return getBasePriceForTier(batch, tier) + getSharingSurcharge(batch, sharing);
}

export function round2(n: number): number {
  return Math.round(n * 100) / 100;
}

export function calculatePricing(input: PricingInput): PricingResult {
  const { rackRate, discountType, discountValue, isGstInclusive, gstRate, isInterState, advancePaid } = input;

  const discountAmount = discountType === 'percentage'
    ? round2(rackRate * (discountValue / 100))
    : round2(discountValue);

  const netBase = round2(Math.max(0, rackRate - discountAmount));

  let taxableBase: number;
  let gstAmount: number;
  let totalPackageCost: number;

  if (isGstInclusive) {
    // GST inclusive: total = netBase, extract base backwards
    // Base = Total / (1 + Rate/100)
    // GST = Total - Base
    const divisor = 1 + gstRate / 100;
    taxableBase = round2(netBase / divisor);
    gstAmount = round2(netBase - taxableBase);
    totalPackageCost = netBase;
  } else {
    // GST exclusive: add GST on top
    taxableBase = netBase;
    gstAmount = round2(netBase * (gstRate / 100));
    totalPackageCost = round2(netBase + gstAmount);
  }

  let cgstAmount = 0;
  let sgstAmount = 0;
  let igstAmount = 0;

  if (isInterState) {
    igstAmount = gstAmount;
  } else {
    cgstAmount = round2(gstAmount / 2);
    sgstAmount = round2(gstAmount - cgstAmount);
  }

  const balanceDue = round2(Math.max(0, totalPackageCost - advancePaid));

  return {
    discountAmount,
    netBase,
    taxableBase,
    gstAmount,
    cgstAmount,
    sgstAmount,
    igstAmount,
    totalPackageCost,
    balanceDue,
  };
}

interface ModalProps {
  open: boolean;
  onClose: () => void;
  title?: string;
  subtitle?: string;
  children: React.ReactNode;
  maxWidth?: string;
}

export function Modal({ open, onClose, title, subtitle, children, maxWidth = 'max-w-lg' }: ModalProps) {
  useEffect(() => {
    if (!open) return;
    const handler = (e: KeyboardEvent) => {
      if (e.key === 'Escape') onClose();
    };
    window.addEventListener('keydown', handler);
    document.body.style.overflow = 'hidden';
    return () => {
      window.removeEventListener('keydown', handler);
      document.body.style.overflow = '';
    };
  }, [open, onClose]);

  if (!open) return null;

  return (
    <div className="fixed inset-0 z-50 flex items-center justify-center p-4">
      <div
        className="absolute inset-0 bg-slate-900/40 backdrop-blur-sm animate-fade-in"
        onClick={onClose}
      />
      <div
        className={`relative w-full ${maxWidth} bg-white rounded-2xl shadow-2xl ring-1 ring-slate-200/60 animate-scale-in max-h-[90vh] overflow-hidden flex flex-col`}
      >
        <div className="flex items-start justify-between px-6 py-5 border-b border-slate-100">
          <div>
            <h2 className="font-display text-lg font-semibold text-slate-900">{title}</h2>
            {subtitle && <p className="mt-0.5 text-sm text-slate-500">{subtitle}</p>}
          </div>
          <button
            onClick={onClose}
            className="p-1.5 rounded-lg text-slate-400 hover:text-slate-600 hover:bg-slate-100 transition-colors"
          >
            <X className="h-5 w-5" />
          </button>
        </div>
        <div className="px-6 py-5 overflow-y-auto scrollbar-thin">{children}</div>
      </div>
    </div>
  );
}

interface ReminderItem {
  type: 'followup' | 'payment' | 'departure';
  label: string;
  detail: string;
  actionLabel: string;
  onAction: () => void;
  urgency: 'overdue' | 'today' | 'soon';

export function RemindersWidget() {
  const [leads, setLeads] = useState<Lead[]>([]);
  const [batches, setBatches] = useState<Batch[]>([]);
  const [passengers, setPassengers] = useState<Passenger[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    async function fetchAll() {
      const [leadsRes, batchesRes, paxRes] = await Promise.all([
        supabase.from('leads').select('*'),
        supabase.from('batches').select('*'),
        supabase.from('passengers').select('*'),
      ]);
      setLeads(leadsRes.data ?? []);
      setBatches(batchesRes.data ?? []);
      setPassengers(paxRes.data ?? []);
      setLoading(false);
    }
    fetchAll();
  }, []);

  const reminders: ReminderItem[] = [];

  // Follow-up reminders
  leads.forEach((lead) => {
    if (!lead.next_follow_up) return;
    const days = daysUntil(lead.next_follow_up);
    if (days <= 0 && lead.status !== 'Converted' && lead.status !== 'Lost') {
      const urgency = isOverdue(lead.next_follow_up) ? 'overdue' : 'today';
      reminders.push({
        type: 'followup',
        label: `Call ${lead.name}`,
        detail: `${lead.destination} · ${urgency === 'overdue' ? 'Overdue' : 'Due today'}${days < 0 ? ` (${Math.abs(days)}d ago)` : ''}`,
        actionLabel: 'Call Now',
        onAction: () => (window.location.href = `tel:${lead.phone.replace(/[^\d+]/g, '')}`),
        urgency,
      });
    }
  });

  // Payment due reminders
  passengers.forEach((pax) => {
    if (!pax.balance_due_date || pax.balance_due <= 0) return;
    const days = daysUntil(pax.balance_due_date);
    if (days <= 7) {
      const batch = batches.find((b) => b.id === pax.batch_id);
      const urgency = isOverdue(pax.balance_due_date) ? 'overdue' : days <= 0 ? 'today' : 'soon';
      reminders.push({
        type: 'payment',
        label: `${pax.name} — ${formatINR(Number(pax.balance_due))} balance`,
        detail: `${batch?.tour_name ?? 'Trip'} · Due ${formatDate(pax.balance_due_date)}`,
        actionLabel: 'WhatsApp',
        onAction: () =>
          openWhatsApp(
            pax.phone || '',
            paymentReminderMessage(pax.name, Number(pax.balance_due), batch?.tour_name ?? 'your trip', pax.balance_due_date!),
          ),
        urgency,
      });
    }
  });

  // Departure briefings
  batches.forEach((batch) => {
    const days = daysUntil(batch.departure_date);
    if (days >= 0 && days <= 2) {
      const urgency: 'today' | 'soon' = days === 0 ? 'today' : 'soon';
      reminders.push({
        type: 'departure',
        label: `${batch.tour_name} departs ${days === 0 ? 'today' : `in ${days}d`}`,
        detail: `${batch.destination} · ${batch.booked_seats}/${batch.total_seats} pax · Send briefing`,
        actionLabel: 'Send Briefing',
        onAction: () => {
          const pax = passengers.find((p) => p.batch_id === batch.id);
          if (pax) {
            openWhatsApp(pax.phone || '', departureBriefingMessage(pax.name, batch.tour_name, batch.departure_date));
          }
        },
        urgency,
      });
    }
  });

  const urgencyOrder = { overdue: 0, today: 1, soon: 2 };
  reminders.sort((a, b) => urgencyOrder[a.urgency] - urgencyOrder[b.urgency]);

  if (loading) {
    return (
      <div className="bg-white rounded-2xl border border-slate-200 p-5 shadow-sm animate-pulse h-32" />
    );
  }

        <p className="text-sm text-slate-400">All caught up — no critical follow-ups today.</p>
      </div>
    );
  }

  const iconForType = {
    followup: <Phone className="h-4 w-4" />,
    payment: <IndianRupee className="h-4 w-4" />,
    departure: <Plane className="h-4 w-4" />,
  };

  const toneForUrgency = {
    overdue: 'border-l-rose-500 bg-rose-50/50',
    today: 'border-l-accent-500 bg-accent-50/50',
    soon: 'border-l-ocean-500 bg-ocean-50/50',
  };

  const iconForUrgency = {
    overdue: 'text-rose-600 bg-rose-100',
    today: 'text-accent-600 bg-accent-100',
    soon: 'text-ocean-600 bg-ocean-100',
  };

  return (
    <div className="bg-white rounded-2xl border border-slate-200 shadow-sm overflow-hidden">

        <span className="ml-auto text-xs font-medium text-slate-400 bg-slate-100 px-2 py-0.5 rounded-full">
          {reminders.length} item{reminders.length !== 1 ? 's' : ''}
        </span>
      </div>
      <div className="divide-y divide-slate-100 max-h-[280px] overflow-y-auto scrollbar-thin">
        {reminders.map((r, i) => (

function colorForDestination(dest: string, batches: Batch[]): number {
  const unique = [...new Set(batches.map((b) => b.destination))];

function formatINRShort(amount: number): string {
  if (amount >= 100000) return `₹${(amount / 100000).toFixed(1)}L`;
  if (amount >= 1000) return `₹${(amount / 1000).toFixed(0)}K`;
  return `₹${amount}`;
}

  const [savingPax, setSavingPax] = useState(false);
  const [roomAssignOpen, setRoomAssignOpen] = useState(false);
  const [assignTarget, setAssignTarget] = useState<Passenger | null>(null);
  const [roomInput, setRoomInput] = useState('');
  const [roomNotesInput, setRoomNotesInput] = useState('');

  const initialPaxForm = {
    name: '', age: '', phone: '', pickup_location: '', emergency_contact: '',
    package_tier: 'Land-to-Land' as PackageTier,
    sharing_type: 'Couple/Double' as SharingType,
    rack_rate: '0', discount_type: 'flat' as DiscountType, discount_value: '0',
    is_gst_inclusive: true, gst_rate: '5', is_inter_state: false,
    advance_paid: '0', advance_date: '', balance_due_date: '',
    payment_mode: 'UPI', utr_number: '',
  };
  const [paxForm, setPaxForm] = useState(initialPaxForm);

  const fetchBatches = useCallback(async () => {
    setLoading(true);
    const { data, error } = await supabase.from('batches').select('*').order('departure_date', { ascending: true });
    if (error) { console.error('Failed to load batches:', error.message); }
    else { setBatches(data ?? []); }
    setLoading(false);
  }, []);

  useEffect(() => { fetchBatches(); }, [fetchBatches]);

  const openBatch = async (batch: Batch) => {
    setSelectedBatch(batch);
    setManifestTab('manifest');
    setPassengerLoading(true);
    const { data, error } = await supabase.from('passengers').select('*').eq('batch_id', batch.id).order('created_at', { ascending: true });
    if (error) { console.error('Failed to load passengers:', error.message); }
    else { setPassengers(data ?? []); }
    setPassengerLoading(false);
  };

  const pricing = useMemo(() => {
    if (!selectedBatch) return null;
    const rack = computeRackRate(selectedBatch, paxForm.package_tier, paxForm.sharing_type);
    return calculatePricing({
      rackRate: paxForm.rack_rate ? parseFloat(paxForm.rack_rate) : rack,
      discountType: paxForm.discount_type,
      discountValue: parseFloat(paxForm.discount_value) || 0,
      isGstInclusive: paxForm.is_gst_inclusive,
      gstRate: parseFloat(paxForm.gst_rate) || 5,
      isInterState: paxForm.is_inter_state,
      advancePaid: parseFloat(paxForm.advance_paid) || 0,
    });
  }, [selectedBatch, paxForm]);

  const exportCSV = () => {
    if (!selectedBatch) return;
    const headers = ['Name', 'Age', 'Phone', 'Package Tier', 'Sharing Type', 'Room', 'Rack Rate', 'Discount', 'Taxable Base', 'CGST', 'SGST', 'IGST', 'Total Cost', 'Advance Paid', 'Balance Due', 'Payment Mode', 'UTR'];
    const rows = passengers.map((p) => [
      p.name, p.age ?? '', p.phone, p.package_tier, p.sharing_type, p.room_number,
      p.rack_rate, p.discount_value, p.taxable_base, p.cgst_amount, p.sgst_amount, p.igst_amount,
      p.total_package_cost, p.advance_paid, p.balance_due, p.payment_mode, p.utr_number,
    ]);
    const csv = [headers, ...rows].map((row) => row.map((cell) => `"${String(cell).replace(/"/g, '""')}"`).join(',')).join('\n');
    const blob = new Blob([csv], { type: 'text/csv' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `manifest-${selectedBatch.tour_name.replace(/\s+/g, '-').toLowerCase()}.csv`;
    a.click();
    URL.revokeObjectURL(url);
  };

  const printManifest = () => {
    if (!selectedBatch) return;
    const rows = passengers.map((p, i) => `
      <tr>
        <td>${i + 1}</td>
        <td class="font-semibold">${p.name}</td>
        <td>${p.phone || '—'}</td>
        <td>${p.age ?? '—'}</td>
        <td>${p.package_tier}</td>
        <td>${p.sharing_type}</td>
        <td>${p.room_number || '—'}</td>
        <td class="text-right">${formatINR(Number(p.total_package_cost))}</td>
        <td class="text-right text-brand font-semibold">${formatINR(Number(p.advance_paid))}</td>
        <td class="text-right text-rose font-semibold">${formatINR(Number(p.balance_due))}</td>
      </tr>`).join('');
    const body = `
      <div class="header">
        <div>
          <div class="company">Trailhead Adventures</div>
          <div class="subtitle">Passenger Manifest</div>
        </div>
        <div class="meta">
          <div>Generated: ${formatDate(new Date().toISOString())}</div>
        </div>
      </div>
      <h1>${selectedBatch.tour_name}</h1>
      <div class="subtitle">
        ${selectedBatch.destination} &middot; Departure: ${formatDate(selectedBatch.departure_date)}
        ${selectedBatch.return_date ? ` &middot; Return: ${formatDate(selectedBatch.return_date)}` : ''}
      </div>
      <div class="summary">
        <div class="summary-card"><div class="label">Passengers</div><div class="value">${passengers.length} / ${selectedBatch.total_seats}</div></div>
        <div class="summary-card"><div class="label">Total Revenue</div><div class="value">${formatINR(totalRevenue)}</div></div>
        <div class="summary-card brand"><div class="label">Advance Collected</div><div class="value">${formatINR(totalAdvance)}</div></div>
        <div class="summary-card rose"><div class="label">Balance Due</div><div class="value">${formatINR(totalBalance)}</div></div>
      </div>
      <table>
        <thead>
          <tr>
            <th>#</th><th>Name</th><th>Phone</th><th>Age</th><th>Package</th><th>Sharing</th><th>Room</th><th class="text-right">Total</th><th class="text-right">Advance</th><th class="text-right">Balance</th>
          </tr>
        </thead>
        <tbody>${rows}</tbody>
      </table>
      <div class="footer">This document was generated by Trailhead CRM on ${formatDate(new Date().toISOString())}.</div>`;
    printDocument(`Manifest - ${selectedBatch.tour_name}`, body);
  };

  const exportRoomingList = () => {
    if (!selectedBatch) return;
    const rooms = getRoomGroups().filter((r) => r.roomNumber);
    const roomCards = rooms.map((r) => `
      <div class="room-card">
        <span class="room-num">${r.roomNumber}</span><span class="room-sharing">${r.sharing}</span>
        <ul>${r.passengers.map((p) => `<li>${p.name}${p.phone ? ' &middot; ' + p.phone : ''}</li>`).join('')}</ul>
      </div>`).join('');
    const unassignedList = unassigned.length > 0
      ? `<div class="unassigned"><h2>Unassigned (${unassigned.length})</h2><ul>${unassigned.map((p) => `<li>${p.name}</li>`).join('')}</ul></div>`
      : '';
    const body = `
      <div class="header">
        <div>
          <div class="company">Trailhead Adventures</div>
          <div class="subtitle">Rooming List</div>
        </div>
        <div class="meta">
          <div>Generated: ${formatDate(new Date().toISOString())}</div>
        </div>
      </div>
      <h1>${selectedBatch.tour_name}</h1>
      <div class="subtitle">${selectedBatch.destination} &middot; Departure: ${formatDate(selectedBatch.departure_date)}</div>
      <div class="summary">
        <div class="summary-card"><div class="label">Rooms Assigned</div><div class="value">${rooms.length}</div></div>
        <div class="summary-card"><div class="label">Guests Assigned</div><div class="value">${passengers.length - unassigned.length}</div></div>
        <div class="summary-card rose"><div class="label">Unassigned</div><div class="value">${unassigned.length}</div></div>
      </div>
      ${roomCards || '<p class="text-slate">No rooms assigned yet.</p>'}
      ${unassignedList}
      <div class="footer">This document was generated by Trailhead CRM on ${formatDate(new Date().toISOString())}.</div>`;
    printDocument(`Rooming - ${selectedBatch.tour_name}`, body);
  };

  const saveTrip = async () => {
    if (!tripForm.tour_name || !tripForm.destination || !tripForm.departure_date) {
      alert('Please fill in tour name, destination, and departure date.');
      return;
    }
    setSavingTrip(true);
    const { error } = await supabase.from('batches').insert({
      tour_name: tripForm.tour_name,
      destination: tripForm.destination,
      departure_date: tripForm.departure_date,
      return_date: tripForm.return_date || null,
      total_seats: parseInt(tripForm.total_seats) || 0,
      booked_seats: 0,
      base_fare: parseFloat(tripForm.base_land_price) || 0,
      gst_rate: parseFloat(tripForm.gst_rate) || 5,
      base_land_price: parseFloat(tripForm.base_land_price) || 0,
      base_train_price: parseFloat(tripForm.base_train_price) || 0,
      base_full_price: parseFloat(tripForm.base_full_price) || 0,
      couple_surcharge: parseFloat(tripForm.couple_surcharge) || 0,
      single_supplement: parseFloat(tripForm.single_supplement) || 0,
    });
    setSavingTrip(false);
    if (error) { alert('Could not create trip. Please try again.'); return; }
    setAddTripOpen(false);
    setTripForm({ tour_name: '', destination: '', departure_date: '', return_date: '', total_seats: '12', gst_rate: '5', base_land_price: '40000', base_train_price: '55000', base_full_price: '70000', couple_surcharge: '3000', single_supplement: '8000' });
    fetchBatches();
  };

  const openAddPax = () => {
    setEditingPax(null);
    const rack = selectedBatch ? computeRackRate(selectedBatch, 'Land-to-Land', 'Couple/Double') : 0;
    setPaxForm({ ...initialPaxForm, rack_rate: String(rack) });
    setAddPaxOpen(true);
  };

  const openEditPax = (p: Passenger) => {
    setEditingPax(p);
    setPaxForm({
      name: p.name, age: p.age ? String(p.age) : '', phone: p.phone, pickup_location: p.pickup_location, emergency_contact: p.emergency_contact,
      package_tier: p.package_tier, sharing_type: p.sharing_type,
      rack_rate: String(p.rack_rate), discount_type: p.discount_type, discount_value: String(p.discount_value),
      is_gst_inclusive: p.is_gst_inclusive, gst_rate: String(p.gst_rate), is_inter_state: Number(p.igst_amount) > 0,
      advance_paid: String(p.advance_paid), advance_date: p.advance_date || '', balance_due_date: p.balance_due_date || '',
      payment_mode: p.payment_mode || 'UPI', utr_number: p.utr_number,
    });
    setAddPaxOpen(true);
  };

  const onTierOrSharingChange = (tier: PackageTier, sharing: SharingType) => {
    if (!selectedBatch || editingPax) {
      setPaxForm((prev) => ({ ...prev, package_tier: tier, sharing_type: sharing }));
      return;
    }
    const rack = computeRackRate(selectedBatch, tier, sharing);
    setPaxForm((prev) => ({ ...prev, package_tier: tier, sharing_type: sharing, rack_rate: String(rack) }));
  };

  const savePassenger = async () => {
    if (!selectedBatch || !pricing) return;
    if (!paxForm.name) { alert('Please enter the passenger name.'); return; }
    setSavingPax(true);
    const payload = {
      batch_id: selectedBatch.id,
      name: paxForm.name,
      age: paxForm.age ? parseInt(paxForm.age) : null,
      phone: paxForm.phone,
      pickup_location: paxForm.pickup_location,
      emergency_contact: paxForm.emergency_contact,
      package_tier: paxForm.package_tier,
      sharing_type: paxForm.sharing_type,
      rack_rate: parseFloat(paxForm.rack_rate) || 0,
      discount_type: paxForm.discount_type,
      discount_value: parseFloat(paxForm.discount_value) || 0,
      is_gst_inclusive: paxForm.is_gst_inclusive,
      gst_rate: parseFloat(paxForm.gst_rate) || 5,
      taxable_base: pricing.taxableBase,
      cgst_amount: pricing.cgstAmount,
      sgst_amount: pricing.sgstAmount,
      igst_amount: pricing.igstAmount,
      total_package_cost: pricing.totalPackageCost,
      total_cost: pricing.totalPackageCost,
      advance_paid: parseFloat(paxForm.advance_paid) || 0,
      advance_date: paxForm.advance_date || null,
      balance_due: pricing.balanceDue,
      balance_due_date: paxForm.balance_due_date || null,
      payment_mode: paxForm.payment_mode,
      utr_number: paxForm.utr_number,
    };
    if (editingPax) {
      const { error } = await supabase.from('passengers').update(payload).eq('id', editingPax.id);
      setSavingPax(false);
      if (error) { alert('Could not update passenger.'); return; }
    } else {
      const { error } = await supabase.from('passengers').insert(payload);
      setSavingPax(false);
      if (error) { alert('Could not add passenger. Please try again.'); return; }
    }
    setAddPaxOpen(false);
    setEditingPax(null);
    setPaxForm(initialPaxForm);
    openBatch(selectedBatch);
    fetchBatches();
  };

  const deletePassenger = async (p: Passenger) => {
    const { error } = await supabase.from('passengers').delete().eq('id', p.id);
    if (error) { alert('Could not remove passenger.'); return; }
    if (selectedBatch) openBatch(selectedBatch);
    fetchBatches();
  };

  const [deleteBatchTarget, setDeleteBatchTarget] = useState<Batch | null>(null);
  const [deletePaxTarget, setDeletePaxTarget] = useState<Passenger | null>(null);

  const deleteBatch = async (batch: Batch) => {
    const { error } = await supabase.from('batches').delete().eq('id', batch.id);
    if (error) { alert('Could not delete trip.'); return; }
    if (selectedBatch?.id === batch.id) setSelectedBatch(null);
    fetchBatches();
  };

  const openRoomAssign = (p: Passenger) => {
    setAssignTarget(p);
    setRoomInput(p.room_number || '');
    setRoomNotesInput(p.room_notes || '');
    setRoomAssignOpen(true);
  };

  const saveRoomAssign = async () => {
    if (!assignTarget) return;
    const { error } = await supabase.from('passengers').update({ room_number: roomInput, room_notes: roomNotesInput }).eq('id', assignTarget.id);
    if (error) { alert('Could not assign room.'); return; }
    setRoomAssignOpen(false);
    setAssignTarget(null);
    if (selectedBatch) openBatch(selectedBatch);
  };

  const unassignRoom = async (p: Passenger) => {
    const { error } = await supabase.from('passengers').update({ room_number: '', room_notes: '' }).eq('id', p.id);
    if (error) { alert('Could not unassign room.'); return; }
    if (selectedBatch) openBatch(selectedBatch);
  };

  const totalAdvance = passengers.reduce((sum, p) => sum + Number(p.advance_paid), 0);
  const totalBalance = passengers.reduce((sum, p) => sum + Number(p.balance_due), 0);
  const totalRevenue = passengers.reduce((sum, p) => sum + Number(p.total_package_cost), 0);

  const getRoomGroups = () => {
    const groups: { roomNumber: string; sharing: string; passengers: Passenger[] }[] = [];
    const roomMap = new Map<string, Passenger[]>();
    passengers.forEach((p) => {
      if (p.room_number) {
        if (!roomMap.has(p.room_number)) roomMap.set(p.room_number, []);
        roomMap.get(p.room_number)!.push(p);
      }
    });
    roomMap.forEach((paxList, roomNumber) => {
      groups.push({ roomNumber, sharing: paxList[0]?.sharing_type || '—', passengers: paxList });
    });
    return groups.sort((a, b) => a.roomNumber.localeCompare(b.roomNumber));
  };

  const roomGroups = getRoomGroups();
  const unassigned = passengers.filter((p) => !p.room_number);

  return (
    <div className="space-y-6">
      <div className="flex justify-end gap-2">
        <button onClick={() => setAddTripOpen(true)} className="inline-flex items-center gap-2 px-4 py-2.5 text-sm font-semibold text-white bg-slate-900 hover:bg-slate-800 rounded-xl transition-colors shadow-sm">
          <Plus className="h-4 w-4" /> Add New Trip
        </button>
      </div>

      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-5">
        {loading && Array.from({ length: 6 }).map((_, i) => (
          <div key={i} className="bg-white rounded-2xl border border-slate-200 p-5 shadow-sm animate-pulse h-44" />
        ))}
        {!loading && batches.map((batch) => {
          const fillPct = batch.total_seats > 0 ? (batch.booked_seats / batch.total_seats) * 100 : 0;
          const remaining = batch.total_seats - batch.booked_seats;
          const days = daysUntil(batch.departure_date);
          return (
            <div key={batch.id} onClick={() => openBatch(batch)} className="group relative text-left bg-white rounded-2xl border border-slate-200 p-5 shadow-sm hover:shadow-md hover:border-brand-300 transition-all duration-200 active:scale-[0.99] md:active:scale-100 cursor-pointer">
              <button onClick={(e) => { e.stopPropagation(); setDeleteBatchTarget(batch); }} className="absolute top-3 right-3 p-1.5 text-rose-500 bg-rose-50 hover:bg-rose-100 rounded-lg transition-colors opacity-0 group-hover:opacity-100" title="Delete trip">
                <Trash2 className="h-3.5 w-3.5" />
              </button>
              <div className="flex items-start justify-between mb-3">
                <div className="flex items-center gap-2">
                  <div className="p-2 rounded-xl bg-ocean-50 text-ocean-600"><Plane className="h-4 w-4" /></div>
                  <span className={`text-xs font-semibold px-2.5 py-1 rounded-full ${days <= 7 ? 'bg-rose-50 text-rose-600' : days <= 14 ? 'bg-accent-50 text-accent-600' : 'bg-brand-50 text-brand-600'}`}>
                    {days > 0 ? `${days}d to go` : 'Departed'}
                  </span>
                </div>
              </div>
              <h3 className="font-display font-semibold text-slate-900 leading-snug mb-1">{batch.tour_name}</h3>
              <div className="flex items-center gap-1.5 text-sm text-slate-500 mb-4">
                <MapPin className="h-3.5 w-3.5" />{batch.destination}
                <span className="mx-1 text-slate-300">|</span>
                <Calendar className="h-3.5 w-3.5" />{formatDate(batch.departure_date)}
              </div>
              <div className="space-y-2">
                <div className="flex items-center justify-between text-sm">
                  <div className="flex items-center gap-1.5 text-slate-600">
                    <Users className="h-4 w-4" />
                    <span className="font-semibold text-slate-900">{batch.booked_seats}</span>
                    <span className="text-slate-400">/ {batch.total_seats} seats</span>
                  </div>
                  <span className="text-xs font-medium text-slate-500">{remaining} left</span>
                </div>
                <div className="h-2 bg-slate-100 rounded-full overflow-hidden">
                  <div className={`h-full rounded-full transition-all duration-500 ${fillPct >= 90 ? 'bg-rose-500' : fillPct >= 70 ? 'bg-accent-500' : 'bg-brand-500'}`} style={{ width: `${fillPct}%` }} />
                </div>
                <div className="flex items-center gap-3 text-xs text-slate-400 pt-1">
                  <span>Land: {formatINR(Number(batch.base_land_price) || Number(batch.base_fare))}</span>
                  {Number(batch.base_train_price) > 0 && <span>Train: {formatINR(Number(batch.base_train_price))}</span>}
                  {Number(batch.base_full_price) > 0 && <span>Full: {formatINR(Number(batch.base_full_price))}</span>}
                </div>
              </div>
            </div>
          );
        })}
      </div>

      {/* Batch Detail Modal with Manifest + Rooming tabs */}
      <Modal open={!!selectedBatch} onClose={() => setSelectedBatch(null)} title={selectedBatch?.tour_name} subtitle={selectedBatch ? `${selectedBatch.destination} · ${formatDate(selectedBatch.departure_date)}` : ''} maxWidth="max-w-5xl">
        {selectedBatch && (
          <div className="space-y-4">
            {/* Summary stats */}
            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
              <div className="bg-slate-50 rounded-xl p-3">
                <div className="text-xs text-slate-500">Booked / Total</div>
                <div className="font-semibold text-slate-900 text-sm mt-0.5">{selectedBatch.booked_seats} / {selectedBatch.total_seats}</div>
              </div>
              <div className="bg-slate-50 rounded-xl p-3">
                <div className="text-xs text-slate-500">Total Revenue</div>
                <div className="font-semibold text-slate-900 text-sm mt-0.5">{formatINR(totalRevenue)}</div>
              </div>
              <div className="bg-brand-50 rounded-xl p-3">
                <div className="text-xs text-brand-600">Advance</div>
                <div className="font-semibold text-brand-700 text-sm mt-0.5">{formatINR(totalAdvance)}</div>
              </div>
              <div className="bg-rose-50 rounded-xl p-3">
                <div className="text-xs text-rose-600">Balance</div>
                <div className="font-semibold text-rose-700 text-sm mt-0.5">{formatINR(totalBalance)}</div>
              </div>
            </div>

            {/* Tab toggle */}
            <div className="flex items-center gap-2 border-b border-slate-200">
              <button onClick={() => setManifestTab('manifest')} className={`inline-flex items-center gap-1.5 px-4 py-2.5 text-sm font-medium border-b-2 transition-colors ${manifestTab === 'manifest' ? 'border-slate-900 text-slate-900' : 'border-transparent text-slate-400 hover:text-slate-600'}`}>
                <List className="h-4 w-4" /> Passenger Manifest
              </button>
              <button onClick={() => setManifestTab('rooming')} className={`inline-flex items-center gap-1.5 px-4 py-2.5 text-sm font-medium border-b-2 transition-colors ${manifestTab === 'rooming' ? 'border-slate-900 text-slate-900' : 'border-transparent text-slate-400 hover:text-slate-600'}`}>
                <BedDouble className="h-4 w-4" /> Rooming Allocator
              </button>
            </div>

            {/* Action buttons */}
            <div className="flex items-center justify-between gap-3 flex-wrap">
              <div className="text-sm text-slate-500">{passengers.length} passenger{passengers.length !== 1 ? 's' : ''}</div>
              <div className="flex gap-2 shrink-0 flex-wrap">
                {manifestTab === 'manifest' ? (
                  <>
                    <button onClick={openAddPax} className="inline-flex items-center gap-1.5 px-3.5 py-2 text-sm font-semibold text-white bg-brand-500 hover:bg-brand-600 rounded-lg transition-colors shadow-sm">
                      <UserPlus className="h-4 w-4" /> Add Passenger
                    </button>
                    <button onClick={exportCSV} className="inline-flex items-center gap-1.5 px-3.5 py-2 text-sm font-medium text-slate-700 bg-slate-100 hover:bg-slate-200 rounded-lg transition-colors">
                      <Download className="h-4 w-4" /> Export CSV
                    </button>
                    <button onClick={printManifest} className="inline-flex items-center gap-1.5 px-3.5 py-2 text-sm font-medium text-white bg-slate-900 hover:bg-slate-800 rounded-lg transition-colors shadow-sm">
                      <Printer className="h-4 w-4" /> Manifest PDF
                    </button>
                  </>
                ) : (
                  <button onClick={exportRoomingList} className="inline-flex items-center gap-1.5 px-3.5 py-2 text-sm font-medium text-white bg-slate-900 hover:bg-slate-800 rounded-lg transition-colors shadow-sm">
                    <Home className="h-4 w-4" /> Rooming PDF
                  </button>
                )}
              </div>
            </div>

            {/* Manifest tab */}
            {manifestTab === 'manifest' && (
              <>
                <div className="hidden md:block border border-slate-200 rounded-xl overflow-hidden">
                  <div className="overflow-x-auto scrollbar-thin max-h-[400px] overflow-y-auto">
                    <table className="w-full">
                      <thead className="sticky top-0">
                        <tr className="border-b border-slate-100 bg-slate-50">
                          <th className="text-left px-4 py-3 text-xs font-semibold text-slate-500 uppercase tracking-wider">Name</th>
                          <th className="text-left px-4 py-3 text-xs font-semibold text-slate-500 uppercase tracking-wider">Package</th>

              )}
              <div className="flex justify-between border-t border-slate-200 pt-1.5"><span className="font-semibold text-slate-900">Total Package Cost</span><span className="font-bold text-slate-900">{formatINR(pricing.totalPackageCost)}</span></div>
            </div>
          )}

          {/* Payment */}
          <div className="border-t border-slate-100 pt-3">
            <div className="text-sm font-semibold text-slate-700 mb-3">Payment</div>
            <div className="grid grid-cols-3 gap-3">
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-1.5">Advance (₹)</label>
                <input type="number" value={paxForm.advance_paid} onChange={(e) => setPaxForm({ ...paxForm, advance_paid: e.target.value })} placeholder="0" className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all" />
              </div>
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-1.5">Advance Date</label>
                <input type="date" value={paxForm.advance_date} onChange={(e) => setPaxForm({ ...paxForm, advance_date: e.target.value })} className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all" />
              </div>
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-1.5">Balance Due Date</label>
                <input type="date" value={paxForm.balance_due_date} onChange={(e) => setPaxForm({ ...paxForm, balance_due_date: e.target.value })} className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all" />
              </div>
            </div>
            <div className="grid grid-cols-2 gap-3 mt-3">
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-1.5">Payment Mode</label>
                <select value={paxForm.payment_mode} onChange={(e) => setPaxForm({ ...paxForm, payment_mode: e.target.value })} className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all">
                  <option value="UPI">UPI</option>
                  <option value="Bank Transfer">Bank Transfer</option>
                  <option value="Cash">Cash</option>
                </select>
              </div>
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-1.5">UTR / Reference No.</label>
                <input type="text" value={paxForm.utr_number} onChange={(e) => setPaxForm({ ...paxForm, utr_number: e.target.value })} placeholder="UTR123456" className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all font-mono" />

            {pricing && (
              <div className="mt-3 flex justify-between bg-rose-50 rounded-xl px-4 py-2.5 text-sm">
                <span className="text-rose-600 font-medium">Balance Due</span>
                <span className="font-bold text-rose-700">{formatINR(pricing.balanceDue)}</span>
              </div>
            )}
          </div>

          <div className="flex justify-end gap-3 pt-2">
            <button onClick={() => { setAddPaxOpen(false); setEditingPax(null); }} className="px-4 py-2.5 text-sm font-medium text-slate-600 hover:bg-slate-100 rounded-xl transition-colors">Cancel</button>
            <button onClick={savePassenger} disabled={savingPax} className="px-5 py-2.5 text-sm font-semibold text-white bg-slate-900 hover:bg-slate-800 rounded-xl transition-colors shadow-sm disabled:opacity-60">{savingPax ? 'Saving...' : editingPax ? 'Update Passenger' : 'Add Passenger'}</button>
          </div>
        </div>
      </Modal>

      {/* Delete Trip Confirmation Dialog */}
      <ConfirmDialog
        open={!!deleteBatchTarget}
        onClose={() => setDeleteBatchTarget(null)}
        onConfirm={() => deleteBatchTarget && deleteBatch(deleteBatchTarget)}
        title="Delete Trip"
        message={`Are you sure you want to delete "${deleteBatchTarget?.tour_name}"? This will also remove all ${'passengers'} on this trip's manifest. This action cannot be undone.`}
        confirmLabel="Delete Trip"
      />

      {/* Delete Passenger Confirmation Dialog */}
      <ConfirmDialog
        open={!!deletePaxTarget}
        onClose={() => setDeletePaxTarget(null)}
        onConfirm={() => deletePaxTarget && deletePassenger(deletePaxTarget)}
        title="Remove Passenger"
        message={`Are you sure you want to remove "${deletePaxTarget?.name}" from this trip? This action cannot be undone.`}
        confirmLabel="Remove Passenger"
      />

      {/* Room Assignment Modal */}
      <Modal open={roomAssignOpen} onClose={() => setRoomAssignOpen(false)} title="Assign Room" subtitle={assignTarget?.name}>
        <div className="space-y-4">
          <div>
            <label className="block text-sm font-medium text-slate-700 mb-1.5">Room / Tent Number</label>
            <input type="text" value={roomInput} onChange={(e) => setRoomInput(e.target.value)} placeholder="e.g. Room 101, Tent 3" className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all" />
          </div>
          <div>
            <label className="block text-sm font-medium text-slate-700 mb-1.5">Room Notes</label>
            <textarea value={roomNotesInput} onChange={(e) => setRoomNotesInput(e.target.value)} rows={2} placeholder="e.g. Twin beds, lake view, ground floor preferred" className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all resize-none" />
          </div>
          <div className="flex justify-end gap-3 pt-2">
            <button onClick={() => setRoomAssignOpen(false)} className="px-4 py-2.5 text-sm font-medium text-slate-600 hover:bg-slate-100 rounded-xl transition-colors">Cancel</button>
            <button onClick={saveRoomAssign} className="px-5 py-2.5 text-sm font-semibold text-white bg-slate-900 hover:bg-slate-800 rounded-xl transition-colors shadow-sm">Assign Room</button>
          </div>

      </Modal>
    </div>

    const seen = new Set<string>();
    const results: ParsedLead[] = [];

    for (const line of lines) {
      const phoneMatch = line.match(phoneRegex);
      if (!phoneMatch) continue;
      let phone = phoneMatch[0].replace(/[\s-]/g, '');
      if (!phone.startsWith('+91') && phone.length === 10) phone = '+91' + phone;
      if (!phone.startsWith('+91') && phone.length === 12) phone = '+' + phone;
      if (seen.has(phone)) continue;
      seen.add(phone);

      const nameMatch = line.match(nameRegex);
      let name = nameMatch?.[1] ?? 'Unknown';
      if (name === 'Unknown') {
        const afterColon = line.split(/[:\]]/).slice(1).join(' ').trim();
        const words = afterColon.split(/\s+/).filter((w) => /^[A-Z][a-z]+$/.test(w));
        if (words.length >= 2) name = words.slice(0, 2).join(' ');
      }

      const destMatch = line.match(destinationKeywords);
      const destination = destMatch?.[0] ?? 'Not specified';

      const dateMatch = line.match(dateRegex);
      let nextFollowUp = '';
      if (dateMatch) {
        const parts = dateMatch[1].split(/[\/.]/);
        if (parts.length === 3) {
          const [d, m, y] = parts;
          const year = y.length === 2 ? '20' + y : y;
          nextFollowUp = `${year}-${m.padStart(2, '0')}-${d.padStart(2, '0')}`;
        }
      }

      const msgText = line.split(/[:\]]/).slice(1).join(' ').trim();
      const notes = msgText.slice(0, 200);

      results.push({
        name,
        phone,
        destination,
        pax: 1,
        notes,
        source: chatText.includes('call') || chatText.includes('Call') ? 'Call Log' : 'WhatsApp',
      });
    }

    setParsedLeads(results);
    setParsing(false);
  };

      source: l.source,

      alert('Could not import leads. Please try again.');
      return;

    setChatImportOpen(false);
    setChatText('');
    setParsedLeads([]);
    fetchLeads();
  };

          <input
            type="text"
            placeholder="Search name, destination, phone..."
            value={search}
            onChange={(e) => setSearch(e.target.value)}
            className="w-full pl-10 pr-4 py-2.5 text-sm bg-white rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all"
          />
        </div>
        <div className="flex items-center gap-2">
          <div className="flex gap-2 overflow-x-auto scrollbar-thin pb-1">
            {(['All', ...STATUS_OPTIONS] as const).map((s) => (
              <button

                  <textarea value={editForm.notes} onChange={(e) => setEditForm({ ...editForm, notes: e.target.value })} rows={2} className="w-full px-3 py-2 text-sm bg-slate-50 rounded-lg border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all resize-none" />
                </div>
                <div className="col-span-2 flex justify-end">
                  <button onClick={saveEdit} disabled={savingEdit} className="inline-flex items-center gap-1.5 px-4 py-2 text-sm font-semibold text-white bg-slate-900 hover:bg-slate-800 rounded-lg transition-colors shadow-sm disabled:opacity-60">
                    <Save className="h-4 w-4" /> {savingEdit ? 'Saving...' : 'Save Changes'}
                  </button>
                </div>
              </div>
            )}

            {!editMode && detailLead.notes && (
              <div className="bg-slate-50 rounded-xl p-3">
                <div className="text-xs text-slate-500 mb-1">Notes</div>
                <div className="text-sm text-slate-700">{detailLead.notes}</div>
              </div>
            )}

            {/* Booked trips */}
            <div>
              <div className="flex items-center gap-2 mb-3">
                <Plane className="h-4 w-4 text-slate-400" />
                <h4 className="text-sm font-semibold text-slate-700">Booked Trips</h4>
                <span className="text-xs text-slate-400">({bookedTrips.length})</span>
              </div>
              {bookedTripsLoading ? (
                <div className="text-center py-4 text-sm text-slate-400">Loading trips...</div>
              ) : bookedTrips.length === 0 ? (
                <div className="text-center py-4 text-sm text-slate-400 bg-slate-50 rounded-xl">No trips booked yet. Convert this lead to add them to a trip.</div>
              ) : (
                <div className="space-y-2">
                  {bookedTrips.map(({ batch, passenger }) => (
                    <div key={passenger.id} className="flex items-center justify-between bg-white border border-slate-200 rounded-xl p-3">
                      <div className="flex items-center gap-3">
                        <div className="p-2 rounded-lg bg-ocean-50 text-ocean-600"><Plane className="h-4 w-4" /></div>
                        <div>
                          <div className="font-medium text-slate-900 text-sm">{batch.tour_name}</div>
                          <div className="text-xs text-slate-400">{batch.destination} · {formatDate(batch.departure_date)}</div>
                        </div>
                      </div>
                      <div className="text-right">
                        <div className="text-xs text-slate-400">{passenger.package_tier}</div>
                        <div className="text-sm font-semibold text-slate-700">{formatINR(Number(passenger.total_package_cost))}</div>
                      </div>
                    </div>
                  ))}
                </div>
              )}
            </div>

            {/* Quick action buttons */}
            <div className="flex gap-2">
              <button onClick={() => handleCall(detailLead)} className="flex-1 inline-flex items-center justify-center gap-2 px-4 py-2.5 text-sm font-semibold text-white bg-slate-900 hover:bg-slate-800 rounded-xl transition-colors shadow-sm">
                <Phone className="h-4 w-4" /> Call Now
              </button>
              <button onClick={() => handleWhatsApp(detailLead)} className="flex-1 inline-flex items-center justify-center gap-2 px-4 py-2.5 text-sm font-semibold text-white bg-brand-500 hover:bg-brand-600 rounded-xl transition-colors shadow-sm">
                <MessageCircle className="h-4 w-4" /> WhatsApp
              </button>
              <button onClick={() => { setDetailLead(null); openConvert(detailLead); }} className="flex-1 inline-flex items-center justify-center gap-2 px-4 py-2.5 text-sm font-semibold text-white bg-brand-600 hover:bg-brand-700 rounded-xl transition-colors shadow-sm">
                <ArrowRightCircle className="h-4 w-4" /> Convert
              </button>
            </div>

            {/* Log new interaction */}
            <div className="border border-slate-200 rounded-xl p-3 space-y-3">
              <div className="text-sm font-semibold text-slate-700">Log Interaction Manually</div>
              <textarea
                value={interactionNote}
                onChange={(e) => setInteractionNote(e.target.value)}
                rows={2}
                placeholder="Add notes about this call or message..."
                className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all resize-none"
              />
              <div className="flex gap-2">
                <button onClick={() => addInteractionNote('Call')} className="flex-1 inline-flex items-center justify-center gap-1.5 px-3 py-2 text-xs font-medium text-slate-700 bg-slate-100 hover:bg-slate-200 rounded-lg transition-colors">
                  <Phone className="h-3.5 w-3.5" /> Log Call
                </button>
                <button onClick={() => addInteractionNote('WhatsApp')} className="flex-1 inline-flex items-center justify-center gap-1.5 px-3 py-2 text-xs font-medium text-white bg-brand-500 hover:bg-brand-600 rounded-lg transition-colors">
                  <MessageCircle className="h-3.5 w-3.5" /> Log WhatsApp
                </button>
                <button onClick={() => addInteractionNote('Note')} className="flex-1 inline-flex items-center justify-center gap-1.5 px-3 py-2 text-xs font-medium text-slate-700 bg-accent-50 hover:bg-accent-100 rounded-lg transition-colors">
                  <StickyNote className="h-3.5 w-3.5" /> Log Note
                </button>
              </div>
            </div>

            {/* Interaction history */}
            <div>
              <div className="flex items-center gap-2 mb-3">
                <History className="h-4 w-4 text-slate-400" />
                <h4 className="text-sm font-semibold text-slate-700">Communication History</h4>
                <span className="text-xs text-slate-400">({interactions.length})</span>
              </div>
              {interactionsLoading ? (
                <div className="text-center py-6 text-sm text-slate-400">Loading history...</div>
              ) : interactions.length === 0 ? (
                <div className="text-center py-6 text-sm text-slate-400 bg-slate-50 rounded-xl">No interactions logged yet. Use Call or WhatsApp buttons to start tracking.</div>
              ) : (
                <div className="space-y-2 max-h-[300px] overflow-y-auto scrollbar-thin">
                  {interactions.map((log) => (
                    <div key={log.id} className="flex items-start gap-3 bg-white border border-slate-200 rounded-xl p-3">
                      <div className={`p-2 rounded-lg shrink-0 ${log.type === 'Call' ? 'bg-slate-100 text-slate-600' : log.type === 'WhatsApp' ? 'bg-brand-50 text-brand-600' : 'bg-accent-50 text-accent-600'}`}>
                        {log.type === 'Call' ? <Phone className="h-3.5 w-3.5" /> : log.type === 'WhatsApp' ? <MessageCircle className="h-3.5 w-3.5" /> : <StickyNote className="h-3.5 w-3.5" />}
                      </div>
                      <div className="min-w-0 flex-1">
                        <div className="flex items-center gap-2">
                          <span className="text-sm font-semibold text-slate-900">{log.type}</span>
                          <span className="inline-flex items-center gap-1 text-xs text-slate-400">
                            <Clock className="h-3 w-3" />
                            {formatDateTime(log.created_at)}
                          </span>
                        </div>
                        {log.notes && (
                          <div className="text-sm text-slate-600 mt-1">{log.notes}</div>
                        )}
                      </div>
                    </div>
                  ))}
                </div>
              )}
            </div>
          </div>
        )}
      </Modal>

      {/* Convert to Passenger Modal */}
      <Modal open={!!convertLead} onClose={() => setConvertLead(null)} title="Convert to Passenger" subtitle={convertLead?.name} maxWidth="max-w-lg">
        {convertLead && (
          <div className="space-y-4">
            <div className="bg-slate-50 rounded-xl p-3 grid grid-cols-2 gap-3 text-sm">
              <div><div className="text-xs text-slate-500">Name</div><div className="font-medium text-slate-900">{convertLead.name}</div></div>
              <div><div className="text-xs text-slate-500">Phone</div><div className="font-medium text-slate-900 font-mono">{convertLead.phone}</div></div>
              <div><div className="text-xs text-slate-500">Destination</div><div className="font-medium text-slate-900">{convertLead.destination}</div></div>
              <div><div className="text-xs text-slate-500">Group Size</div><div className="font-medium text-slate-900">{convertLead.pax} pax</div></div>
            </div>
            <div>
              <label className="block text-sm font-medium text-slate-700 mb-1.5">Select Trip *</label>
              <select value={convertForm.batch_id} onChange={(e) => setConvertForm({ ...convertForm, batch_id: e.target.value })} className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all">
                <option value="">Choose a trip...</option>
                {convertBatches.map((b) => <option key={b.id} value={b.id}>{b.tour_name} — {formatDate(b.departure_date)}</option>)}
              </select>
            </div>
            <div className="grid grid-cols-2 gap-3">
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-1.5">Package Tier</label>
                <select value={convertForm.package_tier} onChange={(e) => setConvertForm({ ...convertForm, package_tier: e.target.value as PackageTier })} className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all">
                  {PACKAGE_TIERS.map((t) => <option key={t} value={t}>{t}</option>)}
                </select>
              </div>
              <div>
                <label className="block text-sm font-medium text-slate-700 mb-1.5">Sharing Type</label>
                <select value={convertForm.sharing_type} onChange={(e) => setConvertForm({ ...convertForm, sharing_type: e.target.value as SharingType })} className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all">
                  {SHARING_TYPES.map((s) => <option key={s} value={s}>{s}</option>)}
                </select>
              </div>
            </div>
            <div>
              <label className="block text-sm font-medium text-slate-700 mb-1.5">Advance Paid (₹)</label>
              <input type="number" value={convertForm.advance_paid} onChange={(e) => setConvertForm({ ...convertForm, advance_paid: e.target.value })} placeholder="0" className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all" />
            </div>
            {convertPricing && (
              <div className="bg-slate-50 rounded-xl p-4 space-y-1.5 text-sm">
                <div className="flex justify-between"><span className="text-slate-500">Package Cost (incl. GST)</span><span className="font-bold text-slate-900">{formatINR(convertPricing.totalPackageCost)}</span></div>
                <div className="flex justify-between"><span className="text-slate-500">Taxable Base</span><span className="text-slate-700">{formatINR(convertPricing.taxableBase)}</span></div>
                {convertPricing.igstAmount > 0 ? (
                  <div className="flex justify-between"><span className="text-slate-500">IGST</span><span className="text-slate-700">{formatINR(convertPricing.igstAmount)}</span></div>
                ) : (
                  <>
                    <div className="flex justify-between"><span className="text-slate-500">CGST</span><span className="text-slate-700">{formatINR(convertPricing.cgstAmount)}</span></div>
                    <div className="flex justify-between"><span className="text-slate-500">SGST</span><span className="text-slate-700">{formatINR(convertPricing.sgstAmount)}</span></div>
                  </>
                )}
                <div className="flex justify-between border-t border-slate-200 pt-1.5"><span className="font-semibold text-rose-600">Balance Due</span><span className="font-bold text-rose-700">{formatINR(convertPricing.balanceDue)}</span></div>
              </div>
            )}
            <div className="flex justify-end gap-3 pt-2">
              <button onClick={() => setConvertLead(null)} className="px-4 py-2.5 text-sm font-medium text-slate-600 hover:bg-slate-100 rounded-xl transition-colors">Cancel</button>
              <button onClick={confirmConvert} disabled={converting || !convertForm.batch_id} className="px-5 py-2.5 text-sm font-semibold text-white bg-brand-600 hover:bg-brand-700 rounded-xl transition-colors shadow-sm disabled:opacity-60">{converting ? 'Converting...' : 'Convert to Passenger'}</button>
            </div>
          </div>
        )}
      </Modal>

      {/* Call Note Modal */}
      <Modal open={!!noteModalLead} onClose={() => setNoteModalLead(null)} title="Add Call Note" subtitle={noteModalLead ? noteModalLead.name : ''}>
        <div className="space-y-5">
          <div>
            <label className="block text-sm font-medium text-slate-700 mb-1.5">Discussion Notes</label>
            <textarea value={noteText} onChange={(e) => setNoteText(e.target.value)} rows={5} placeholder="What was discussed? Any special requests, concerns, or commitments?" className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all resize-none" />
          </div>
          <div>
            <label className="block text-sm font-medium text-slate-700 mb-1.5">Next Follow-up Date</label>
            <div className="relative">
              <Calendar className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-slate-400" />
              <input type="date" value={followUpDate} onChange={(e) => setFollowUpDate(e.target.value)} className="w-full pl-10 pr-4 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all" />
            </div>
          </div>
          <div className="flex justify-end gap-3 pt-2">
            <button onClick={() => setNoteModalLead(null)} className="px-4 py-2.5 text-sm font-medium text-slate-600 hover:bg-slate-100 rounded-xl transition-colors">Cancel</button>
            <button onClick={saveNote} disabled={saving} className="px-5 py-2.5 text-sm font-semibold text-white bg-slate-900 hover:bg-slate-800 rounded-xl transition-colors shadow-sm disabled:opacity-60">{saving ? 'Saving...' : 'Save Note'}</button>
          </div>
        </div>
      </Modal>

      {/* Add Lead Modal */}
      <Modal open={addLeadOpen} onClose={() => setAddLeadOpen(false)} title="Add New Lead" subtitle="Manually add a customer inquiry">
        <div className="space-y-4">
          <div>
            <label className="block text-sm font-medium text-slate-700 mb-1.5">Name *</label>
            <input type="text" value={leadForm.name} onChange={(e) => setLeadForm({ ...leadForm, name: e.target.value })} placeholder="Customer name" className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all" />
          </div>
          <div>
            <label className="block text-sm font-medium text-slate-700 mb-1.5">Phone *</label>
            <input type="tel" value={leadForm.phone} onChange={(e) => setLeadForm({ ...leadForm, phone: e.target.value })} placeholder="+91XXXXXXXXXX" className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all font-mono" />
          </div>
          <div className="grid grid-cols-2 gap-3">
            <div>
              <label className="block text-sm font-medium text-slate-700 mb-1.5">Destination</label>
              <input type="text" value={leadForm.destination} onChange={(e) => setLeadForm({ ...leadForm, destination: e.target.value })} placeholder="e.g. Spiti Valley" className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all" />
            </div>
            <div>
              <label className="block text-sm font-medium text-slate-700 mb-1.5">Group Size</label>
              <input type="number" value={leadForm.pax} onChange={(e) => setLeadForm({ ...leadForm, pax: e.target.value })} className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all" />
            </div>
          </div>
          <div>
            <label className="block text-sm font-medium text-slate-700 mb-1.5">Follow-up Date</label>
            <div className="relative">
              <Calendar className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-slate-400" />
              <input type="date" value={leadForm.next_follow_up} onChange={(e) => setLeadForm({ ...leadForm, next_follow_up: e.target.value })} className="w-full pl-10 pr-4 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all" />
            </div>
          </div>
          <div className="flex justify-end gap-3 pt-2">
            <button onClick={() => setAddLeadOpen(false)} className="px-4 py-2.5 text-sm font-medium text-slate-600 hover:bg-slate-100 rounded-xl transition-colors">Cancel</button>
            <button onClick={saveLead} disabled={savingLead} className="px-5 py-2.5 text-sm font-semibold text-white bg-slate-900 hover:bg-slate-800 rounded-xl transition-colors shadow-sm disabled:opacity-60">{savingLead ? 'Adding...' : 'Add Lead'}</button>
          </div>
        </div>
      </Modal>

      {/* Import CSV Modal */}
      <Modal open={importOpen} onClose={() => { setImportOpen(false); setImportData([]); }} title="Import Leads from CSV" subtitle="Bulk upload customer inquiries">
        <div className="space-y-4">
          {importData.length === 0 ? (
            <>
              <p className="text-sm text-slate-500">Upload a CSV file with columns: <span className="font-mono text-xs bg-slate-100 px-1.5 py-0.5 rounded">Name, Phone, Destination, Group Size, Advance Paid, Follow-up Date</span></p>
              <div className="flex flex-col gap-3">
                <input ref={fileInputRef} type="file" accept=".csv" onChange={handleCsvUpload} className="hidden" />
                <button onClick={() => fileInputRef.current?.click()} className="w-full flex flex-col items-center gap-2 py-8 border-2 border-dashed border-slate-200 rounded-xl text-slate-400 hover:border-brand-300 hover:text-brand-500 transition-colors">
                  <Upload className="h-8 w-8" />
                  <span className="text-sm font-medium">Choose CSV File</span>
                </button>
                <button onClick={downloadSampleCsv} className="inline-flex items-center justify-center gap-2 px-4 py-2.5 text-sm font-medium text-ocean-600 bg-ocean-50 hover:bg-ocean-100 rounded-xl transition-colors">
                  <Download className="h-4 w-4" />
                  Download Sample Template
                </button>
              </div>
            </>
          ) : (
            <>
              <div className="flex items-center justify-between">
                <p className="text-sm font-medium text-slate-700">{importData.length} leads ready to import</p>
                <button onClick={() => { setImportData([]); if (fileInputRef.current) fileInputRef.current.value = ''; }} className="p-1.5 rounded-lg text-slate-400 hover:text-slate-600 hover:bg-slate-100 transition-colors">
                  <X className="h-4 w-4" />
                </button>
              </div>
              <div className="max-h-[300px] overflow-y-auto scrollbar-thin border border-slate-200 rounded-xl">
                <table className="w-full">
                  <thead className="sticky top-0 bg-slate-50">
                    <tr>
                      <th className="text-left px-3 py-2 text-xs font-semibold text-slate-500">Name</th>
                      <th className="text-left px-3 py-2 text-xs font-semibold text-slate-500">Phone</th>
                      <th className="text-left px-3 py-2 text-xs font-semibold text-slate-500">Destination</th>
                      <th className="text-center px-3 py-2 text-xs font-semibold text-slate-500">Pax</th>
                    </tr>
                  </thead>
                  <tbody className="divide-y divide-slate-100">
                    {importData.map((l, i) => (
                      <tr key={i}>
                        <td className="px-3 py-2 text-sm text-slate-700">{l.name}</td>
                        <td className="px-3 py-2 text-sm text-slate-600 font-mono">{l.phone}</td>
                        <td className="px-3 py-2 text-sm text-slate-600">{l.destination}</td>
                        <td className="px-3 py-2 text-sm text-slate-600 text-center">{l.pax}</td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              </div>
              <div className="flex justify-end gap-3">
                <button onClick={() => { setImportData([]); if (fileInputRef.current) fileInputRef.current.value = ''; }} className="px-4 py-2.5 text-sm font-medium text-slate-600 hover:bg-slate-100 rounded-xl transition-colors">Cancel</button>
                <button onClick={confirmImport} disabled={importing} className="px-5 py-2.5 text-sm font-semibold text-white bg-slate-900 hover:bg-slate-800 rounded-xl transition-colors shadow-sm disabled:opacity-60">{importing ? 'Importing...' : `Import ${importData.length} Leads`}</button>
              </div>
            </>
          )}
        </div>
      </Modal>

      {/* Import from Chat / Log Modal */}
      <Modal open={chatImportOpen} onClose={() => { setChatImportOpen(false); setParsedLeads([]); setChatText(''); }} title="Import from WhatsApp Chat / Call Log" subtitle="Paste raw chat export or call log text" maxWidth="max-w-2xl">
        <div className="space-y-4">
          {parsedLeads.length === 0 ? (
            <>
              <div className="flex items-start gap-2 bg-accent-50 rounded-xl p-3 text-sm text-accent-700">
                <AlertCircle className="h-4 w-4 shrink-0 mt-0.5" />
                <p>Paste raw WhatsApp chat export or phone call log text below. Phone numbers (e.g. +91XXXXXXXXXX), names, and dates will be automatically detected.</p>
              </div>
              <textarea
                value={chatText}
                onChange={(e) => setChatText(e.target.value)}
                rows={10}
                placeholder={`Paste WhatsApp chat or call log here...\n\nExample:\n[01/09/26, 10:30 AM] Rahul Sharma: Hi, I'm interested in the Manali trip\n[01/09/26, 11:15 AM] +91 98765 43210: What are the dates?\n[02/09/26, 09:00 AM] Priya Nair: Looking for a Spiti Valley package\n+91 98123 45678 called at 14:30`}
                className="w-full px-3.5 py-2.5 text-sm bg-slate-50 rounded-xl border border-slate-200 focus:outline-none focus:ring-2 focus:ring-brand-500/20 focus:border-brand-400 transition-all resize-none font-mono"
              />
              <div className="flex justify-end gap-3">
                <button onClick={() => setChatImportOpen(false)} className="px-4 py-2.5 text-sm font-medium text-slate-600 hover:bg-slate-100 rounded-xl transition-colors">Cancel</button>
                <button onClick={parseChatText} disabled={!chatText.trim() || parsing} className="inline-flex items-center gap-2 px-5 py-2.5 text-sm font-semibold text-white bg-brand-600 hover:bg-brand-700 rounded-xl transition-colors shadow-sm disabled:opacity-60">
                  <ClipboardPaste className="h-4 w-4" /> {parsing ? 'Parsing...' : 'Parse Text'}
                </button>
              </div>
            </>
          ) : (
            <>
              <div className="flex items-center justify-between">
                <p className="text-sm font-medium text-slate-700">{parsedLeads.length} contact{parsedLeads.length !== 1 ? 's' : ''} detected</p>
                <button onClick={() => { setParsedLeads([]); setChatText(''); }} className="p-1.5 rounded-lg text-slate-400 hover:text-slate-600 hover:bg-slate-100 transition-colors">
                  <X className="h-4 w-4" />
                </button>
              </div>
              <div className="max-h-[350px] overflow-y-auto scrollbar-thin border border-slate-200 rounded-xl">
                <table className="w-full">
                  <thead className="sticky top-0 bg-slate-50">
                    <tr>
                      <th className="text-left px-3 py-2 text-xs font-semibold text-slate-500">Name</th>
                      <th className="text-left px-3 py-2 text-xs font-semibold text-slate-500">Phone</th>
                      <th className="text-left px-3 py-2 text-xs font-semibold text-slate-500">Destination</th>
                      <th className="text-left px-3 py-2 text-xs font-semibold text-slate-500">Source</th>
                    </tr>
                  </thead>
                  <tbody className="divide-y divide-slate-100">
                    {parsedLeads.map((l, i) => (
                      <tr key={i}>
                        <td className="px-3 py-2 text-sm text-slate-700">{l.name}</td>
                        <td className="px-3 py-2 text-sm text-slate-600 font-mono">{l.phone}</td>
                        <td className="px-3 py-2 text-sm text-slate-600">{l.destination}</td>
                        <td className="px-3 py-2 text-sm text-slate-500">{l.source}</td>
                      </tr>
                    ))}
                  </tbody>
                </table>
              </div>
              <div className="flex justify-end gap-3">
                <button onClick={() => { setParsedLeads([]); setChatText(''); }} className="px-4 py-2.5 text-sm font-medium text-slate-600 hover:bg-slate-100 rounded-xl transition-colors">Start Over</button>
                <button onClick={importParsedLeads} disabled={importingParsed} className="px-5 py-2.5 text-sm font-semibold text-white bg-slate-900 hover:bg-slate-800 rounded-xl transition-colors shadow-sm disabled:opacity-60">{importingParsed ? 'Importing...' : `Import ${parsedLeads.length} Lead${parsedLeads.length !== 1 ? 's' : ''}`}</button>
              </div>
            </>
          )}
        </div>
      </Modal>

      {/* Delete Confirmation Dialog */}
      <ConfirmDialog
        open={!!deleteTarget}
        onClose={() => setDeleteTarget(null)}
        onConfirm={confirmDeleteLead}
        title="Delete Lead"
        message={`Are you sure you want to delete "${deleteTarget?.name}"? This will also remove all interaction history for this lead. This action cannot be undone.`}
        confirmLabel="Delete Lead"
      />
    </div>
  );
}

function StatusSelect({ lead, onChange }: { lead: Lead; onChange: (lead: Lead, status: LeadStatus) => void }) {
  return (
    <div className="relative inline-block">
      <select value={lead.status} onChange={(e) => onChange(lead, e.target.value as LeadStatus)} className={`appearance-none pl-3 pr-8 py-1 text-xs font-semibold rounded-full ring-1 cursor-pointer transition-all ${statusStyles[lead.status]}`}>
        {STATUS_OPTIONS.map((s) => (<option key={s} value={s}>{s}</option>))}
      </select>
      <span className={`absolute left-2 top-1/2 -translate-y-1/2 h-1.5 w-1.5 rounded-full ${statusDot[lead.status]}`} />
      <ChevronRight className="absolute right-1.5 top-1/2 -translate-y-1/2 h-3 w-3 rotate-90 pointer-events-none opacity-50" />
    </div>
  );
}

function StatusBadge({ status }: { status: LeadStatus }) {
  return (

}

function StatCard({ label, value, icon, tone }: { label: string; value: number; icon: React.ReactNode; tone: 'ocean' | 'slate' | 'accent' | 'brand' }) {
  const toneMap = { ocean: 'bg-ocean-50 text-ocean-600', slate: 'bg-slate-100 text-slate-600', accent: 'bg-accent-50 text-accent-600', brand: 'bg-brand-50 text-brand-600' };
  return (
    <div className="bg-white rounded-2xl border border-slate-200 p-4 shadow-sm">
      <div className="flex items-center gap-3">
        <div className={`p-2.5 rounded-xl ${toneMap[tone]}`}>{icon}</div>
        <div>
          <div className="text-2xl font-display font-bold text-slate-900 leading-none">{value}</div>
          <div className="text-xs text-slate-500 mt-1">{label}</div>
        </div>
      </div>
    </div>
  );
}
